cmake_minimum_required(VERSION 3.10)
project(cenatMD LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")

find_package(MPI REQUIRED)
find_package(OpenMP REQUIRED)

# NOTA: esta lista de fuentes es PROVISIONAL (solo Tarea 1 de Persona A).
# En la Tarea 6 se completa con src/particle.cpp, src/physics.cpp,
# src/init.cpp, src/io.cpp a medida que Persona B y A los entreguen.
add_executable(cenatMD
    src/main.cpp
    src/mpi_utils.cpp
)

target_include_directories(cenatMD PRIVATE include)
target_link_libraries(cenatMD PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX)
